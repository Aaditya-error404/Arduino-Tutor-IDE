2. COMPLETE WORKING & RUNTIME EXECUTION PIPELINE Lexer • Virtual Clock • Hardware State • 3D Kinematics 
flowchart TD
    subgraph S1["1. Code Ingestion"]
        C1["Student writes Arduino C++ in sketch.ino"] --> C2["handleCodeChange hoists code to setCustomCode"]
        C2 --> C3["Student clicks [RUN SIM] or presses Space"]
    end

    subgraph S2["2. AST Lexer & Compiler (compiler.ts)"]
        C3 --> COMP1["Resolves constants (LED_BUILTIN=13, A0-A5=14-19)"]
        COMP1 --> COMP2["Balanced parenthesis parser (setupSteps[] & loopSteps[])"]
        COMP2 --> COMP3["Phase 0 variable & array hoisting into vars{}"]
        COMP3 --> COMP4["Generates CompiledProgram AST"]
    end

    subgraph S3["3. Virtual Clock Stepper (virtualClock.ts)"]
        COMP4 --> CLK1["globalVirtualClock.start()"]
        CLK1 --> CLK2["Ticks deltaMs (10ms steps @ 1x-10x speed)"]
        CLK2 --> CLK3["Dispatches onClockTick(deltaMs, virtualTime)"]
    end

    subgraph S4["4. Hardware State Machine (useWorkbenchStore.ts)"]
        CLK3 --> HW1{"Remaining Delay <= 0?"}
        HW1 -- No --> HW2["Deduct deltaMs • Advance ScopeBuffer"]
        HW1 -- Yes --> HW3["Increment steps • Watchdog Check (>10k steps)"]
        HW3 --> HW4["Execute InstructionNode"]

        HW4 --> HW_DISPATCH{"Instruction"}
        HW_DISPATCH -- pinMode --> ACT_PIN["Update pinStates & pinModes"]
        HW_DISPATCH -- digitalWrite --> ACT_DIG["Set Pin HIGH/LOW (LED, Relay, Motors)"]
        HW_DISPATCH -- analogWrite --> ACT_PWM["Set pwmStates (Pins 3, 5, 6, 9, 10, 11)"]
        HW_DISPATCH -- tone / noTone --> ACT_TONE["Set buzzerActive & buzzerToneFreq"]
        HW_DISPATCH -- delay --> ACT_DELAY["Set stepRemainingDelayMs"]
        HW_DISPATCH -- Serial.print --> ACT_SER["Buffer string in serialLogs[]"]

        ACT_PIN --> HW_SYNC["Mutate HardwareState Store"]
        ACT_DIG --> HW_SYNC
        ACT_PWM --> HW_SYNC
        ACT_TONE --> HW_SYNC
        ACT_DELAY --> HW_SYNC
        ACT_SER --> HW_SYNC
    end
